<h1 className="h1">а <em>как в TypeScript?</em></h1>

<div className="code code-sm">
```typescript
// TS НЕ поддерживает async get/set напрямую
// Но можно типизировать через Promise:
interface AsyncField {
    get(): Promise<number>;
    set(value: number): Promise<void>;
}
// Или через прокси-объект:
interface AsyncObject {
    field: Promise<number>;
}
// Присваивание = number, не Promise!
const value: number = await (myObj.field = 321);
// type error: assignment возвращает 321, не Promise
```
</div>
