<h1 className="h1"><em>POJO</em> с прототипом</h1>

<div className="code">
```javascript
const someObject = {
    name: 'myType',
    createdAt: new Date()
};
Object.setPrototypeOf(someObject, {
    describe () {
        return `Type ${this.name} created at ${this.createdAt}`;
    }
});
console.log(someObject.describe());  // работает!
```
</div>

<div className="callout">
  <div className="lead"><code className="ic">POJO</code> + <code className="ic">setPrototypeOf</code> === <code className="ic">new Constructor</code></div>
</div>

-----

<h1 className="h1"><em>Используя</em> Constructor</h1>

<div className="code">
```javascript
// Constructor — это Interface
function Interface(name) {
    this.name = name;
    this.createdAt = new Date();
}
Interface.prototype.describe = function() {
    return `Interface ${this.name} created at ${this.createdAt}`;
};
const myInstance = new Interface('MyInterface');
console.log(myInstance.describe());  // тот же результат!
```
</div>

<div className="callout">
  <div className="lead">Прото-<span className="pink">Тип</span> получил <span className="pink">Интерфейс!</span></div>
</div>

<h1 className="h1">а <em>как в TypeScript?</em></h1>

<div className="code">
```typescript
type MyType = {
    name: string;
    createdAt: Date;
};
interface MyInterface {
    describe(): string;
}
type MyInstance = MyType & MyInterface;
interface MyCtor extends NewableFunction {
    new (name: string): MyInstance;
    prototype: MyInterface;
}
```
</div>

<h1 className="h1 lower" style={{fontSize:'54px'}}>что такое <code className="ic" style={{fontSize:'0.85em'}}>NewableFunction</code>?</h1>

<div className="code code-md">
```typescript
// Встроенный тип TS — НЕ для прямого использования!
interface NewableFunction {
    new (...args: any[]): any;
    prototype: any;
}

// НЕЛЬЗЯ: просто присвоить — потеряем прототип
const ctor: NewableFunction = function() {};
// ctor.prototype — any, не расширяется!
```
</div>

<div className="callout pink">
  <div className="lead"><code className="ic">NewableFunction</code> — <em>интерфейс-заглушка</em>,<br/> не тип для конструкторов</div>
</div>

-----

<h1 className="h1 lower" style={{fontSize:'54px'}}>логично через <code className="ic" style={{fontSize:'0.85em'}}>extends</code></h1>

<div className="code code-md">
```typescript
// Declaration merging + extends = работающий прототип
interface MyConstructor extends NewableFunction {
    new (): MyInstance;
    prototype: MyProto;
}
interface MyProto {
    shared: string;
}
interface MyInstance extends MyProto {
    own: number;
}
```
</div>

<div className="callout">
  <div className="lead"><code className="ic">extends</code> активирует declaration merging для <span className="pink">прототипа</span></div>
</div>
