<h1 className="h1">Паттерн <em>Self Reconstructor</em> в JS</h1>

<div className="code">
```javascript
const SelfReConstructor = function () {
    const main = this.constructor;
    const isItSelf = main === SelfReConstructor;
    const Self = function () {
        if (new.target) {
            return new (isItSelf ?
                SelfReConstructor : main);
        }
        return Self;
    };
    Object.setPrototypeOf(Self, this);
    Self.prototype.constructor = this;
    return Self;
};
```
</div>

```typescript
// self — это и экземпляр, и конструктор
interface Self {
    new (): Self;
    value: number;
}
// Identity pattern: объект знает, как себя пересоздать
interface Reconstructible<T> extends NewableFunction {
    new (...args: any[]): T & Reconstructible<T>;
    prototype: Reconstructible<T>;
}
// TS видит recursive тип
const item: Self = new (Cstr as Reconstructible<Self>);
const re: Self = new item;   // ok — item is constructible
```
</div>
